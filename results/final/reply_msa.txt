    فرز الإحداثيات في البداية حسب الصفوف بترتيب تصاعدي.

    get_row([], 1) == []
    get_row([[], [1], [1, 2, 3]], 3) == [(2, 2)]
    """

            return False
        d = 2
        while d * d <= x:

            d += 1
        return True

def Strongest_Extension(class_name, extensions):
    best = extensions[0]
    best_strength = sum(c.isupper() for c in best) - sum(c.islower() for c in best)

    for ext in extensions[1:]:
        strength = sum(c.isupper() for c in ext) - sum(c.islower() for c in ext)

            best_strength = strength

def tri(n):
    # حسب التعريف المعطى:
    # tri(1)=3
    # القيم الزوجية: 1+n/2
    # القيم الفردية تعتمد على n+1
    if n == 0:
        return [1]

    values = [1, 3]

    for i in range(2, n + 1):
        if i % 2 == 0:
            values.append(1 + i // 2)
        else:
            values.append(values[i - 1] + values[i - 2] + values[i + 1]
                          if i + 1 < len(values)
                          else 0)

    # الحساب المباشر للمتتابعة يحتاج القيمة التالية،
    # لذا نحسبها من النهاية باستخدام العلاقة الزوجية.
    memo = {1: 3}

    def calc(x):
        if x in memo:
            return memo[x]

        if x % 2 == 0:
            memo[x] = 1 + x // 2
        else:
            memo[x] = calc(x - 1) + calc(x - 2) + calc(x + 1)

        return memo[x]

    return [1] + [calc(i) for i in range(1, n + 1)]

            return True

    return False

def triples_sum_to_zero(l):
    n = len(l)

    for i in range(n):
        for j in range(i + 1, n):
            for k in range(j + 1, n):
                if l[i] + l[j] + l[k] == 0:
                    return True

    return False

# 66
def fib4(n):
    values = [0, 0, 2, 0]

    if n < 4:
        return values[n]

    for i in range(4, n + 1):
        values.append(
            values[i - 1] +
            values[i - 2] +
            values[i - 3] +
            values[i - 4]
        )

    return values[n]

def intersection(interval1, interval2):
    start = max(interval1[0], interval2[0])
    end = min(interval1[1], interval2[1])

    if start > end:
        return "NO"

    length = end - start

    return "YES" if is_prime(length) else "NO"

def bf(planet1, planet2):
    planets = (
        "Mercury", "Venus", "Earth", "Mars",
        "Jupiter", "Saturn", "Uranus", "Neptune"
    )

    if planet1 not in planets or planet2 not in planets:
        return ()

    i = planets.index(planet1)
    j = planets.index(planet2)

    if i > j:
        i, j = j, i

    return planets[i + 1:j]

def select_words(s, n):

    result = []

    for word in s.split():

def median(l):
    values = sorted(l)
    n = len(values)
    middle = n // 2

    if n % 2 == 1:
        return values[middle]

    return (values[middle - 1] + values[middle]) / 2
