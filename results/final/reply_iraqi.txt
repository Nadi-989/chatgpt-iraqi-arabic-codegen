def tri(n): 
    """كلنا نعرف متسلسلة فيبوناتشي، وعلماء الرياضيات درسوها هواية بالقرون اللي فاتت. بس اللي الناس ما يعرفوه هي متسلسلة تريبوناتشي. تريبوناتشي تنعرف هيج: 
tri(1) = 3 
tri(n) = 1 + n / 2، إذا n زوجي. 
tri(n) = tri(n - 1) + tri(n - 2) + tri(n + 1)، إذا n فردي. 
مثلاً: 
tri(2) = 1 + (2 / 2) = 2 
tri(4) = 3 
tri(3) = tri(2) + tri(1) + tri(4) 
= 2 + 3 + 3 = 8 
ينطوك رقم صحيح مو سالب n، ولازم ترجّع ليستة بيها أول n + 1 أرقام من متسلسلة تريبوناتشي. 
أمثلة: 
tri(3) = [1, 3, 2, 8] 
    """

from typing import List

def rescale_to_unit(numbers: List[float]) -> List[float]:
    mn = min(numbers)
    mx = max(numbers)
    if mn == mx:
        return [0.0] * len(numbers)
    return [(x - mn) / (mx - mn) for x in numbers]

    balance = 0
    nested = False

    for ch in string:
        if ch == "[":
            balance += 1
            if balance >= 2:
                nested = True
        elif ch == "]":

def Strongest_Extension(class_name, extensions):
    strongest = extensions[0]
    best_strength = sum(c.isupper() for c in strongest) - sum(
        c.islower() for c in strongest
    )

    for ext in extensions[1:]:
        strength = sum(c.isupper() for c in ext) - sum(c.islower() for c in ext)
        if strength > best_strength:
            best_strength = strength
            strongest = ext

    return f"{class_name}.{strongest}"

def tri(n):
    values = [0] * (n + 1)

    if n >= 0:
        values[0] = 1
    if n >= 1:
        values[1] = 3
    if n >= 2:
        values[2] = 2

    for i in range(3, n + 1):
        if i % 2 == 0:
            values[i] = 1 + i // 2
        else:
            # Because the definition uses tri(n + 1),
            # calculate the needed even term first.
            values[i] = values[i - 1] + values[i - 2] + (1 + (i + 1) // 2)

    return values

    if not l:
        return None

    values = sorted(l)
    n = len(values)

    if n % 2 == 1:
        return values[n // 2]

    return (values[n // 2 - 1] + values[n // 2]) / 2
