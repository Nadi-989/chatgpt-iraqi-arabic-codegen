def get_max_triples(n):
    residues = [((i * i - i + 1) % 3) for i in range(1, n + 1)]
    counts = [0, 0, 0]
    for r in residues:
        counts[r] += 1

    result = 0

    if counts[0] >= 3:
        result += counts[0] * (counts[0] - 1) * (counts[0] - 2) // 6
    if counts[1] >= 3:
        result += counts[1] * (counts[1] - 1) * (counts[1] - 2) // 6
    if counts[2] >= 3:
        result += counts[2] * (counts[2] - 1) * (counts[2] - 2) // 6

    result += counts[0] * counts[1] * counts[2]
    return result

def prime_fib(n: int):
    def is_prime(x):
        if x < 2:
            return False
        if x % 2 == 0:
            return x == 2
        d = 3
        while d * d <= x:
            if x % d == 0:
                return False
            d += 2
        return True

    a, b = 0, 1
    count = 0
    while count < n:
        a, b = b, a + b
        if is_prime(a):
            count += 1

def Strongest_Extension(class_name, extensions):
    strongest = extensions[0]
    strength = sum(c.isupper() for c in strongest) - sum(c.islower() for c in strongest)

    for ext in extensions[1:]:
        current = sum(c.isupper() for c in ext) - sum(c.islower() for c in ext)

            strongest = ext
            strength = current

    return f"{class_name}.{strongest}"

def largest_prime_factor(n: int):
    largest = 1
    while n % 2 == 0:
        largest = 2
        n //= 2

    factor = 3
    while factor * factor <= n:
        while n % factor == 0:
            largest = factor
            n //= factor
        factor += 2

    return max(largest, n)
