from typing import List

def rescale_to_unit(numbers: List[float]) -> List[float]:
    mn, mx = min(numbers), max(numbers)
    if mx == mn:
        return [0.0 for _ in numbers]
    return [(x - mn) / (mx - mn) for x in numbers]

    for i in range(1, n + 1):
        counts[(i * i - i + 1) % 3] += 1

    total = 0
    total += counts[0] * (counts[0] - 1) * (counts[0] - 2) // 6
    total += counts[1] * (counts[1] - 1) * (counts[1] - 2) // 6
    total += counts[2] * (counts[2] - 1) * (counts[2] - 2) // 6
    total += counts[0] * counts[1] * counts[2]
    total += counts[0] * counts[1] * (counts[1] - 1) // 2

    def is_prime(x):
        if x < 2:
            return False
        d = 2
        while d * d <= x:
            if x % d == 0:
                return False
            d += 1
        return True

    a, b = 0, 1
    count = 0
    while count < n:
        a, b = b, a + b
        if is_prime(a):
            count += 1
    return a

def median(l: list):
    values = sorted(l)
    n = len(values)
    mid = n // 2
    if n % 2:
        return values[mid]
